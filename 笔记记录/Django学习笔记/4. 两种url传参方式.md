## 1. 第一种url传参给函数

```python
from app02 import models

# views.py

def depart_list(request):
    """ 部门表"""
    from django.shortcuts import render
    
    
    queryset = models.Department.objects.all() # 获取部门表所有数据
    # print(queryset)   # <QuerySet [<Department: Department object (1)>, <Department: Department object (2)>]>
    # print(type(queryset)) # <class 'django.db.models.query.QuerySet'>
    # for obj in queryset:
    #     print(obj.title)
                                             # 把这个queryset传给html
    return render(request, "depart_list.html", {"queryset":queryset})
```

                                                        |                   |                           
                                                        |     views.py      |
                                                        |                   |
                                                        |      传参给        |
                                                        |                   |
                                                        | depart_list.html  |
                                                        |                   |
```html

<!-- depart_list.html-->

{% for obj in queryset %}
<tr>
    {% csrf_token %}
    <th scope="row">{{ obj.id }}</th>
    <td>{{ obj.title }}</td>
    <td>
        <a class="btn btn-danger btn-sm" href="/depart/delete/?nid={{ obj.id }}">删 除</a>
    </td>
</tr>
{% endfor %}

```
