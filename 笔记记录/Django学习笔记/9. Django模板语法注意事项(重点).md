法2使用动态传参

```python
from django.db import models

class UserInfo(models.Model):
    
# models.py 中

    sex_choice = (
        (1, "男"),
        (0, "女"),
    )
         
    sex = models.SmallIntegerField(verbose_name="性别", choices=sex_choice)
      
```
```python

# views.py文件中
from app02 import models
from django.shortcuts import render

def user_add(request):
    """ 新建用户 """
    # 提前定义一个字典,再传给模板文件 {((1, "男"),(0, "女"),)}
    context = {
        'sex_choice': models.UserInfo.sex_choice, #这是一个元组数据,用数字取值
    }

    return render(request, 'user_add.html', context) # 把context传给模板文件


```
```html

<!-- user_add.html 模板文件中-->

<select class="form-control">
    {% for item in sex_choice %}
    <option value="{{ item.0 }}">{{ item.1 }}</option>
    {% endfor %}
</select>

```
