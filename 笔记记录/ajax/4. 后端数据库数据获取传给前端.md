# 想要去数据库获取数据时：对象/字典

```python


    # values("title", "price", "status", "oid")  # 字典，｛“title”:xx,"price":xx｝
row_object = models.Order.objects.filter(id=uid).values("title","price","status").first()


```

```python
对象
# queryset = [obj1]
queryset = models.Order.objects.filter()
字典
# queryset = ｛"id":1,“title”:xxx,"price":20｝
queryset = models.Order.objects.filter().values("id","title", "price")

多个对象
# queryset = [obj1,obj2,obj3,..] 数据库每个对象封装成了字典
queryset = models.Order.objects.all()
字典列表
# queryset = [｛"id":1,“title”:xxx,"price":20｝]
queryset = models.Order.objects.all().values("id","title", "price")

使用 values_list 从数据库可以取出的类型为：元组列表
# queryset = [("id":1),(“title”:xxx),("price":20)]
queryset = models.Order.objects.all().values_list("id", "title","price")

```
